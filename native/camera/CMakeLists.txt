cmake_minimum_required(VERSION 3.24)
project(SaiphCamera LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(SaiphCamera SHARED ${GENERATED}/UnityCaptureFilter.cpp ${GENERATED}/streams.cpp producer.cpp registration.cpp ${GENERATED}/SaiphCamera.def)
target_include_directories(SaiphCamera PRIVATE ${GENERATED})
target_compile_definitions(SaiphCamera PRIVATE NDEBUG _WINDOWS _USRDLL)
target_compile_options(SaiphCamera PRIVATE /O2 /W3 /GS /guard:cf)
target_link_options(SaiphCamera PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(SaiphCamera PRIVATE ole32 oleaut32 advapi32 user32 gdi32 winmm strmiids uuid)
add_executable(SaiphCameraTest test.cpp)
target_include_directories(SaiphCameraTest PRIVATE ${GENERATED})
target_link_libraries(SaiphCameraTest PRIVATE ole32 oleaut32 advapi32 strmiids uuid)
